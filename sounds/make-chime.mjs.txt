// A small, sweet "done" chime: two soft bell notes a fifth apart, no noise,
// gentle attack, quiet peak. About 1.4 s.
import { writeFileSync } from 'node:fs'
const SR = 44100, DUR = 1.4, N = Math.floor(SR * DUR)
const out = new Float32Array(N)
// a soft bell: pure fundamental plus a faint octave, slow-ish decay, 12 ms fade-in
const bell = (start, f, amp, decay) => {
  for (let i = Math.floor(start * SR); i < N; i++) {
    const t = i / SR - start
    const att = Math.min(1, t / 0.012)
    const env = att * Math.exp(-t * decay)
    out[i] += amp * env * (Math.sin(2 * Math.PI * f * t) + 0.12 * Math.sin(2 * Math.PI * 2 * f * t))
  }
}
bell(0.0, 1318.5, 1.0, 4.2)   // E6
bell(0.13, 1975.5, 0.7, 4.8)  // B6
// fade out the tail
for (let i = 0; i < N; i++) { const t = i / SR; if (t > DUR - 0.25) out[i] *= (DUR - t) / 0.25 }
let peak = 0; for (const v of out) peak = Math.max(peak, Math.abs(v))
const TARGET = 0.11 // about -19 dBFS: soft
const buf = Buffer.alloc(44 + N * 2)
buf.write('RIFF', 0); buf.writeUInt32LE(36 + N * 2, 4); buf.write('WAVEfmt ', 8)
buf.writeUInt32LE(16, 16); buf.writeUInt16LE(1, 20); buf.writeUInt16LE(1, 22)
buf.writeUInt32LE(SR, 24); buf.writeUInt32LE(SR * 2, 28); buf.writeUInt16LE(2, 32); buf.writeUInt16LE(16, 34)
buf.write('data', 36); buf.writeUInt32LE(N * 2, 40)
for (let i = 0; i < N; i++) buf.writeInt16LE(Math.round((out[i] / peak) * TARGET * 32767), 44 + i * 2)
writeFileSync(process.argv[2], buf)
console.log('wrote', buf.length, 'bytes, peak', TARGET)
